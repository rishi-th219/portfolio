Query Performance

Partitioned by calendar date with indexes on Date_Key and Facility_Key so reports load quickly.

MEASURE // ATTENDANCE RATE DAX 
Compliance_Rate_% = 
VAR TotalScheduled = COUNTROWS(Fact_Attendance_Daily)
VAR VerifiedPresent = 
    CALCULATE(
        COUNTROWS(Fact_Attendance_Daily),
        Fact_Attendance_Daily[Is_Present] = 1
    )
RETURN
    DIVIDE(VerifiedPresent, TotalScheduled, 0)
MEASURE // ABSENCE RISK TIER DAX 
Defaulter_Risk_Tier = 
VAR ConsecutiveUnexcused = [Rolling_Consecutive_Absences]
RETURN
    SWITCH(
        TRUE(),
        ConsecutiveUnexcused >= 3, "High Risk Tier 1",
        ConsecutiveUnexcused >= 1, "Moderate Risk Tier 2",
        "Compliant"
    )
